(function () { const isRajWatchTab = window.location.search.includes("rajwatch=1") || localStorage.getItem("rw_auto_stream") === "1"; if (window.location.search.includes("rajwatch=1")) { localStorage.setItem("rw_auto_stream", "1"); } // Screen ke bottom-right mein chhota sa live status badge const badge = document.createElement("div"); badge.style.cssText = "position:fixed;bottom:16px;right:16px;z-index:99999;background:#0f172a;color:#34d399;" + "border:2px solid #10b981;padding:8px 14px;border-radius:10px;font:bold 12px system-ui;" + "box-shadow:0 4px 20px rgba(0,0,0,0.5);cursor:pointer;"; badge.textContent = "RajWatch 𝕏 Bridge: Starting..."; document.body.appendChild(badge); let streaming = isRajWatchTab; badge.onclick = () => { streaming = !streaming; localStorage.setItem("rw_auto_stream", streaming ? "RajWatch 𝕏 Bridge: ON" : "RajWatch 𝕏 Bridge: PAUSED (Click to ON)"; }; function grabCount(article, testId) { const el = article.querySelector(`[data-testid="${testId}"]`); if (!el) return "0"; const txt = (el.innerText || "").trim(); return txt || "0"; } function scrapeAndPush() { if (!streaming) return; const articles = document.querySelectorAll('article[data-testid="tweet"]'); const tweets = []; articles.forEach(art => { try { const timeEl = art.querySelector("time"); if (!timeEl) return; const linkEl = timeEl.closest("a"); if (!linkEl) return; const href = linkEl.getAttribute("href") || ""; const m = href.match(/^\/([^/]+)\/status\/(\d+)/); if (!m) return; const handle = m[1]; const id = m[2]; const textEl = art.querySelector('[data-testid="tweetText"]'); const text = textEl ? textEl.innerText.trim() : ""; if (!text) return; const nameEl = art.querySelector('[data-testid="User-Name"] span'); const name = nameEl ? nameEl.innerText.trim() : handle; tweets.push({ id, handle, name, text, url: `https://x.com/${handle}/status/${id}`, created_at: timeEl.getAttribute("datetime") || new Date().toISOString(), likes: grabCount(art, "like"), retweets: grabCount(art, "retweet"), replies: grabCount(art, "reply") }); } catch (e) {} }); if (tweets.length > 0) { chrome.runtime.sendMessage( { type: "PUSH_TWEETS", tweets, query: window.location.search }, res => { if (chrome.runtime.lastError || !res || !res.ok) { badge.textContent = "RajWatch 𝕏 Bridge: Render waking up..."; } else { badge.textContent = `● RajWatch 𝕏 LIVE (${res.data.total_stored} tweets synced)`; } } ); } } // Har 6 second mein tweets scan karke Render par bhejo setInterval(scrapeAndPush, 6000); setTimeout(scrapeAndPush, 2500); // Har 45 second mein page auto-refresh taaki ekdum naye live tweets aate rahein setInterval(() => { if (streaming) window.location.reload(); }, 45000); })();
